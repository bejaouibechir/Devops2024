## Exemple de pipeline 

```groovy

// Définit le pipeline déclaratif piloté par Jenkins.
pipeline {
    // Exécute le pipeline sur un agent Jenkins disponible et éligible.
    agent any

    // Déclare les variables d'environnement disponibles dans cette portée du pipeline.
    environment {
        REMOTE_HOST = 'IP_PUBLIQUE_VM'
        REMOTE_USER = 'UTILISATEUR_SSH'
        SSH_KEY = '/chemin/vers/id_rsa'
        SCRIPT_NAME = 'infos.sh'
        DEST_PATH = '/tmp/infos.sh'
    }

    // Organise les phases du pipeline ; leur ordre est séquentiel hors branches parallèles.
    stages {
        // Ouvre la phase « Créer script bash local », visible séparément dans le suivi du build.
        stage('Créer script bash local') {
            // Regroupe les actions exécutées dans cette phase.
            steps {
                // Génère ce fichier dans le workspace ; le texte multiligne ci-dessous constitue son contenu.
                writeFile file: "${env.SCRIPT_NAME}", text: '''#!/bin/bash
# Évalue date sur la machine d'exécution puis affiche la date.
echo "Date : $(date)"
# Évalue whoami sur la machine d'exécution puis affiche le compte courant.
echo "Utilisateur : $(whoami)"
# Évalue hostname sur la machine d'exécution puis affiche le nom de la machine.
echo "Machine : $(hostname)"
'''
                // Écrit un message dans la console Jenkins ; ce message seul ne réalise pas l'opération annoncée.
                echo "Script ${env.SCRIPT_NAME} généré localement."
            }
        }

        // Ouvre la phase « Transférer script vers machine distante », visible séparément dans le suivi du build.
        stage('Transférer script vers machine distante') {
            // Regroupe les actions exécutées dans cette phase.
            steps {
                // Transfère le fichier local vers la machine décrite par remote.
                sshPut remote: [
                    host: "${env.REMOTE_HOST}",
                    user: "${env.REMOTE_USER}",
                    identityFile: "${env.SSH_KEY}",
                    // Cet exemple accepte toute clé d'hôte ; il ne vérifie donc pas l'identité du serveur.
                    allowAnyHosts: true
                ],
                from: "${env.SCRIPT_NAME}",
                into: "/tmp"
                // Écrit un message dans la console Jenkins ; ce message seul ne réalise pas l'opération annoncée.
                echo "Script transféré dans /tmp sur la machine distante."
            }
        }

        // Ouvre la phase « Donner permissions et exécuter le script », visible séparément dans le suivi du build.
        stage('Donner permissions et exécuter le script') {
            // Regroupe les actions exécutées dans cette phase.
            steps {
                // Exécute la commande indiquée sur la machine distante via SSH Pipeline Steps.
                sshCommand remote: [
                    host: "${env.REMOTE_HOST}",
                    user: "${env.REMOTE_USER}",
                    identityFile: "${env.SSH_KEY}",
                    // Cet exemple accepte toute clé d'hôte ; il ne vérifie donc pas l'identité du serveur.
                    allowAnyHosts: true
                ],
                command: """
                    # Applique les permissions indiquées au fichier ou au répertoire avant son utilisation.
                    chmod +x ${env.DEST_PATH} && \
                    ${env.DEST_PATH}
                """
            }
        }
    }
}


```
