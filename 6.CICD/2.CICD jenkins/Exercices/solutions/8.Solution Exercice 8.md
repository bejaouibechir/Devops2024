## Exemple de prototype

```groovy

// Définit le pipeline déclaratif piloté par Jenkins.
pipeline {
    // Exécute le pipeline sur un agent Jenkins disponible et éligible.
    agent any

    // Déclare les variables d'environnement disponibles dans cette portée du pipeline.
    environment {
        REMOTE_HOST = 'IP_PUBLIQUE_VM'
        REMOTE_USER = 'UTILISATEUR_SSH'
        INVENTORY = 'inventory.ini'
        PLAYBOOK = 'infos.yml'
        SSH_KEY = '/chemin/vers/id_rsa'
    }

    // Organise les phases du pipeline ; leur ordre est séquentiel hors branches parallèles.
    stages {
        // Ouvre la phase « Créer fichier d’inventaire », visible séparément dans le suivi du build.
        stage('Créer fichier d’inventaire') {
            // Regroupe les actions exécutées dans cette phase.
            steps {
                // Génère ce fichier dans le workspace ; le texte multiligne ci-dessous constitue son contenu.
                writeFile file: "${env.INVENTORY}", text: """
[remote]
${env.REMOTE_HOST} ansible_user=${env.REMOTE_USER} ansible_ssh_private_key_file=${env.SSH_KEY} ansible_ssh_common_args='-o StrictHostKeyChecking=no'
"""
                // Écrit un message dans la console Jenkins ; ce message seul ne réalise pas l'opération annoncée.
                echo "Inventaire généré"
            }
        }

        // Ouvre la phase « Créer le playbook Ansible », visible séparément dans le suivi du build.
        stage('Créer le playbook Ansible') {
            // Regroupe les actions exécutées dans cette phase.
            steps {
                // Génère ce fichier dans le workspace ; le texte multiligne ci-dessous constitue son contenu.
                writeFile file: "${env.PLAYBOOK}", text: '''
---
# Sélectionne le groupe d'hôtes défini dans l'inventaire.
- hosts: remote
  # Active ou désactive la collecte automatique des informations système au début du play.
  gather_facts: no
  # Exécute ces tâches dans l'ordre sur les hôtes sélectionnés.
  tasks:
    - name: Afficher la date
      # Exécute cette commande au moyen du shell de la machine cible.
      ansible.builtin.shell: date

    - name: Afficher l’utilisateur courant
      # Exécute cette commande au moyen du shell de la machine cible.
      ansible.builtin.shell: whoami

    - name: Afficher le nom de la machine
      # Exécute cette commande au moyen du shell de la machine cible.
      ansible.builtin.shell: hostname
'''
                // Écrit un message dans la console Jenkins ; ce message seul ne réalise pas l'opération annoncée.
                echo "Playbook Ansible généré"
            }
        }

        // Ouvre la phase « Exécuter le playbook », visible séparément dans le suivi du build.
        stage('Exécuter le playbook') {
            // Regroupe les actions exécutées dans cette phase.
            steps {
                // Exécute le playbook sur les hôtes de l'inventaire avec les options d'authentification indiquées.
                sh "ansible-playbook -i ${env.INVENTORY} ${env.PLAYBOOK}"
            }
        }
    }
}


```
